1. Bubble Sort
procedure bubbleSort(A : list of sortable items)

    n = length(A)

    repeat
        swapped = false

        for i = 1 to n - 1 inclusive do

            if A[i-1] > A[i] then
                temp = A[i-1]
                A[i-1] = A[i]
                A[i] = temp

                swapped = true
            end if

        end for

        n = n - 1

    until not swapped

end procedure

Idea: Compare adjacent elements and move the largest element toward the end.

2. Selection Sort
procedure selectionSort(A : list of sortable items)

    n = length(A)

    for i = 0 to n - 2 inclusive do

        minIndex = i

        for j = i + 1 to n - 1 inclusive do

            if A[j] < A[minIndex] then
                minIndex = j
            end if

        end for

        temp = A[i]
        A[i] = A[minIndex]
        A[minIndex] = temp

    end for

end procedure

Idea: Find the smallest element from the unsorted part and place it at the current position.

3. Insertion Sort
procedure insertionSort(A : list of sortable items)

    n = length(A)

    for i = 1 to n - 1 inclusive do

        key = A[i]
        j = i - 1

        while j >= 0 and A[j] > key do
            A[j + 1] = A[j]
            j = j - 1
        end while

        A[j + 1] = key

    end for

end procedure

Idea: Take one element and insert it into its correct position in the already sorted left part.

4. Quick Sort

Quick Sort has two parts: quickSort and partition.

Partition
procedure partition(A, low, high)

    pivot = A[high]

    i = low - 1

    for j = low to high - 1 inclusive do

        if A[j] < pivot then

            i = i + 1

            temp = A[i]
            A[i] = A[j]
            A[j] = temp

        end if

    end for

    temp = A[i + 1]
    A[i + 1] = A[high]
    A[high] = temp

    return i + 1

end procedure
Quick Sort
procedure quickSort(A, low, high)

    if low < high then

        pivotIndex = partition(A, low, high)

        quickSort(A, low, pivotIndex - 1)

        quickSort(A, pivotIndex + 1, high)

    end if

end procedure

Idea: Choose a pivot, place smaller elements before it and larger elements after it, then sort both parts recursively.

5. Merge Sort

Merge Sort also has two parts: mergeSort and merge.

Merge
procedure merge(A, low, mid, high)

    create temporary list B

    i = low
    j = mid + 1
    k = 0

    while i <= mid and j <= high do

        if A[i] <= A[j] then
            B[k] = A[i]
            i = i + 1
        else
            B[k] = A[j]
            j = j + 1
        end if

        k = k + 1

    end while

    while i <= mid do
        B[k] = A[i]
        i = i + 1
        k = k + 1
    end while

    while j <= high do
        B[k] = A[j]
        j = j + 1
        k = k + 1
    end while

    for k = 0 to length(B) - 1 do
        A[low + k] = B[k]
    end for

end procedure
Merge Sort
procedure mergeSort(A, low, high)

    if low < high then

        mid = (low + high) / 2

        mergeSort(A, low, mid)

        mergeSort(A, mid + 1, high)

        merge(A, low, mid, high)

    end if

end procedure

Idea: Divide the array into smaller parts, sort them recursively, and merge the sorted parts.

Quick revision
Bubble
→ Compare adjacent elements
→ Largest moves to end

Selection
→ Find minimum
→ Put minimum at current position

Insertion
→ Take one element
→ Insert into sorted left part

Quick
→ Choose pivot
→ Partition
→ Recursively sort both sides

Merge
→ Divide into halves
→ Recursively sort
→ Merge sorted halves

For Quick Sort and Merge Sort, remember: both are Divide and Conquer, but Quick Sort divides using a pivot/partition, while Merge Sort divides by halves and combines using merge.
